// neoforge.mods.toml from its template, with the build's properties
var generateModMetadata = tasks.register("generateModMetadata", ProcessResources) {
    var replaceProperties = [
            minecraft_version      : minecraft_version,
            minecraft_version_range: minecraft_version_range,
            neo_version            : neo_version,
            loader_version_range   : loader_version_range,
            mod_id                 : mod_id,
            mod_name               : mod_name,
            mod_license            : mod_license,
            mod_version            : mod_version,
            mod_authors            : mod_authors,
    ]
    inputs.properties replaceProperties
    expand replaceProperties
    from "src/main/templates"
    into "build/generated/sources/modMetadata"
}
sourceSets.main.resources.srcDir generateModMetadata
neoForge.ideSyncTask generateModMetadata

// The engine stands alone: its sources may use only its own packages (dev.lull, dev.lab), Minecraft, NeoForge and
// libraries - never a mod built on it.
def engineSources = fileTree('src/main/java')
def checkEngine = tasks.register('checkEngineIndependent') {
    inputs.files(engineSources)
    doLast {
        def bad = engineSources.files.findAll { f ->
            f.name.endsWith('.java') && f.text.find(/(?m)^\s*import\s+(static\s+)?dev\.(?!lull\.|lab\.|engine_room\.flywheel\.)/)
        }
        if (!bad.isEmpty()) throw new GradleException("the engine imports another mod's package: " + bad*.name)
    }
}
tasks.named('compileJava') { dependsOn checkEngine }

// the dedicated server reads its console from this build's standard input: tools/*.sh drive it with commands
tasks.named('runServer') { standardInput = System.in }

// extra JVM arguments for the runs, from the tools: -PrunJvmArgs="-Dlab.opt.defaults=true -Dsleep.seedEntities=true"
// -PrunArgs="--width 1280 --height 720": program arguments for the dev runs (tools/client_bench.py)
if (project.hasProperty('runArgs')) {
    def extra = project.property('runArgs').toString().trim()
    if (!extra.isEmpty()) neoForge.runs.configureEach { programArguments.addAll(extra.split(/\s+/) as List) }
}
if (project.hasProperty('runJvmArgs')) {
    def extra = project.property('runJvmArgs').toString().trim()
    if (!extra.isEmpty()) neoForge.runs.configureEach { jvmArguments.addAll(extra.split(/\s+/) as List) }
}
